[lua-script mob/oracle/probe.lua]
function oncmd()
local found, e, cmd, what = strfind(argument, "^(%a+) (%a+)")
if (cmd ~= "say") then
return FALSE
end
if (what == "npc") then
say("isnpc me=" .. tostring(isnpc(me)) .. " ch=" .. tostring(isnpc(ch)))
elseif (what == "round") then
say("round " .. round(2.7) .. " " .. round(-2.7) .. " " .. round("3.9"))
elseif (what == "skip") then
say("skip [" .. skip_spaces("   hi  there ") .. "]")
elseif (what == "world") then
local m = inworld("mob", 12220)
local c = inworld("char", ch.name)
say("inworld mob=" .. m.name .. " char=" .. c.name .. " none=" .. tostring(inworld("mob", 1)) .. " nobody=" .. tostring(inworld("char", "nobody")))
elseif (what == "see") then
say("cansee ch=" .. tostring(cansee(ch)))
elseif (what == "fight") then
say("isfighting ch=" .. tostring(isfighting(ch)) .. " me=" .. tostring(isfighting(me)))
elseif (what == "dir") then
say("direction n=" .. direction(12296, 12281) .. " far=" .. direction(12296, 12297) .. " same=" .. direction(12296, 12296))
elseif (what == "obj") then
local o = ch.objs[1]
say("obj " .. o.name .. " corpse=" .. tostring(iscorpse(o)) .. " canget=" .. tostring(canget(o)) .. " shop=" .. tostring(item_check(o)))
else
return FALSE
end
return TRUE
end
