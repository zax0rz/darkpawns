function oncmd()
local found, e, cmd, what = strfind(argument, "^(%a+) (%a+)")
if (cmd ~= "say") then
return FALSE
end
if (what == "object") then
local o = me.objs[1]
spell(nil, o, 29, 1)
say("object invisible=" .. tostring(obj_flagged(o, 5)))
spell(nil, o, 3, 0)
say("object blessed=" .. tostring(obj_flagged(o, 8)))
else
return FALSE
end
return TRUE
end
