[lua-script mob/oracle/probe.lua]
function oncmd()
local found, e, cmd, what = strfind(argument, "^(%a+) (%a+)")
if (cmd ~= "say") then
return FALSE
end
if (what == "echoroom") then
echo(room, "room", "The ground trembles.\r\n")
elseif (what == "echolocal") then
echo(ch, "local", "A voice echoes.")
elseif (what == "echozone") then
echo(ch, "zone", "A zone-wide rumble.\r\n")
elseif (what == "gossip") then
gossip("Hear ye, hear ye!")
elseif (what == "social") then
social(ch, "bow")
elseif (what == "follow") then
local r = follow(ch, 0)
say("follow returned " .. type(r))
elseif (what == "hunt") then
set_hunt(me, ch)
local a = tostring(ishunt(me))
set_hunt(me, nil)
say("hunt set=" .. a .. " cleared=" .. tostring(ishunt(me)))
elseif (what == "stand") then
action(me, "stand")
elseif (what == "sanct") then
spell(ch, nil, SPELL_SANCTUARY, 0)
elseif (what == "cure") then
local r = spell(ch, nil, SPELL_CURE_LIGHT, 1)
say("spell returned " .. type(r))
elseif (what == "unaffect") then
unaffect(ch)
say("unaffected")
else
return FALSE
end
return TRUE
end
