[lua-script mob/oracle/probe.lua]
function oncmd()
local found, e, cmd, what = strfind(argument, "^(%a+) (%a+)")
if (cmd ~= "say") then
return FALSE
end
if (what == "emote") then
emote("smiles warmly.")
elseif (what == "number") then
say("numbers " .. number(1, 100) .. " " .. number(1, 6) .. " " .. number(5, 5))
elseif (what == "outdoor") then
echo(ch, "outdoor", "Thunder rolls across the sky.\r\n")
elseif (what == "global") then
echo(ch, "global", "The heavens rumble.")
elseif (what == "saveobj") then
local o = ch.objs[1]
o.name = "a polished dragon's tooth"
o.cost = 5
o.weight = 0
save_obj(o)
say("saved")
elseif (what == "kill") then
local m = mload(12204, 12296)
raw_kill(m, me, TYPE_UNDEFINED)
say("killed")
elseif (what == "tport") then
tport(ch, 12281)
else
return FALSE
end
return TRUE
end
