[lua-script mob/oracle/dispel_good.lua]
function oncmd()
if (argument == "say shield") then
aff_flags(ch, "set", 7)
aff_flags(ch, "set", 12)
return TRUE
end
if (argument == "say state") then
say("sanctuary=" .. tostring(aff_flagged(ch, 7)) .. " protect-evil=" .. tostring(aff_flagged(ch, 12)))
return TRUE
end
return FALSE
end
