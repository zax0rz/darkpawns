# Live registered do_bite -> damage() -> SkillMessage set-1015 miss proof.
# A production room oncmd sets the actor's valid PLR_WEREWOLF/AFF_WEREWOLF
# state and the victim's Sanctuary/Protection from Good affects. That makes the
# registered Bite damage path reduce the attacker's 12 base damage to zero.
# depth-case: bite.skill-message-miss
# depth-case: fightmsg.150.record-1

[fixture]
empty-players
quiet-mobs
spawn-mob 12220 999 8105 122
mob-script 12220 oracle/wolf.lua 512

[lua-script mob/oracle/wolf.lua]
function oncmd()
if (argument == "say shield") then
aff_flags(ch, "set", 7)
aff_flags(ch, "set", 13)
return TRUE
end
if (argument == "say state") then
say("sanctuary=" .. tostring(aff_flagged(ch, 7)) .. " protect-good=" .. tostring(aff_flagged(ch, 13)))
return TRUE
end
if (argument ~= "say moon") then return FALSE end
plr_flags(ch, "set", 16)
aff_flags(ch, "set", 27)
return TRUE
end
