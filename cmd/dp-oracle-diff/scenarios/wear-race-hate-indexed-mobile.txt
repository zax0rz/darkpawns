# Stock object 5200 applies races 18 and 20. After the real wield command,
# indexed Race-18 mobile #5502 must select the wearer through APPLY_RACE_HATE.
# #8552 is in unindexed 85.mob and is not loaded by C. The shared C affect /
# equipment path is src/handler.c:205-238; the pulse consumer is
# src/mobact.c:236-258. Its onpulse_pc fixture marks the wearer stunned and
# puts the mobile on a violence wait, then returns FALSE so the later
# race-hate scan still runs without adding a player counterattack round.
# depth-case: wear.player-race-hate-mobile-dispatch

[fixture]

[lua-script mob/oracle/race-hate-isolate.lua]
function onpulse_pc()
me.timer = 2
for i=1,10 do
if room.char[i] and room.char[i].name == "Racehatepeer" then
room.char[i].pos = 3
save_char(room.char[i])
end
end
save_char(me)
return FALSE
end
