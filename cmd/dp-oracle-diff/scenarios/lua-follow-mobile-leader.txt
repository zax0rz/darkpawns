function oncmd()
local found, e, cmd, what = strfind(argument, "^(%a+) (%a+)")
if (cmd == "say" and what == "followmob") then
follow(inworld("mob", 4818), 0)
say("follow mob complete")
return TRUE
end
return FALSE
end
