[lua-script mob/oracle/flamestrike_matrix.lua]
function oncmd()
if (argument ~= "say flame") then
return FALSE
end

aff_flags(inworld("char", "DrownDeath"), "set", 31)
aff_flags(inworld("char", "DrownGod"), "set", 31)
return TRUE
end
