# onpulse Lua paths. Room 3065's oncmd writes the peer's exact position; room
# 22500's onpulse invokes lua_spell with that PC as me even when normal command
# input cannot run, then restores standing. The oncmd spell probes bypass
# do_cast's earlier self and class gates and reach cast_spell directly.

[lua-script room/30/pattern_3065.lua]
function oncmd()
local found, e, cmd, what = strfind(argument, "^(%a+) (%a+)")
if (cmd ~= "say") then
return FALSE
end
if (what == "sleeping") then
ch.pos = 4
elseif (what == "resting") then
ch.pos = 5
elseif (what == "sitting") then
ch.pos = 6
elseif (what == "fighting") then
ch.pos = 7
elseif (what == "default") then
ch.pos = 3
elseif (what == "selfonly") then
spell(room.char[1], nil, SPELL_INVULNERABILITY, 1)
elseif (what == "notself") then
spell(ch, nil, SPELL_BLINDNESS, 1)
elseif (what == "group") then
spell(nil, nil, 48, 1)
else
return FALSE
end
return TRUE
end

[lua-script room/30/pattern_dmg.lua]
function onpulse()
if (me.pos == 4) then
spell(nil, nil, SPELL_SANCTUARY, 1)
ch.pos = 8
elseif (me.pos == 5) then
spell(nil, nil, SPELL_SANCTUARY, 1)
ch.pos = 8
elseif (me.pos == 6) then
spell(nil, nil, SPELL_SANCTUARY, 1)
ch.pos = 8
elseif (me.pos == 7) then
spell(nil, nil, SPELL_SANCTUARY, 1)
ch.pos = 8
elseif (me.pos == 3) then
spell(nil, nil, SPELL_SANCTUARY, 1)
ch.pos = 8
end
end
