[lua-script mob/oracle/luna_cast_gates.lua]
function oncmd()
  if (argument == "say master") then
    follow(ch, 1)
    spell(ch, nil, SPELL_SANCTUARY, 1)
    return TRUE
  end
  return FALSE
end
