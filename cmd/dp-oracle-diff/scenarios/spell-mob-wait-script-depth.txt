# src/spell_parser.c:845-848; src/act.wizard.c:1175-1200.

[lua-script mob/oracle/mob_wait.lua]
function oncmd()
if (argument == "say wait") then
me.timer = 1000000
save_char(me)
return TRUE
elseif (argument == "say cast") then
if (me.timer > 0) then
tell("Setpeer", "Timer state: " .. me.timer)
else
tell("Setpeer", "Timer state: zero")
end
spell(me, nil, 1, 1)
return TRUE
end
return FALSE
end
