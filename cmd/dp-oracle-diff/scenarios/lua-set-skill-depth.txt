function oncmd()
if (argument == "say valid") then
local r = set_skill(ch, 134, 37)
if (type(r) == "userdata" and r == ch.struct) then
say("valid player struct")
else
say("valid player result " .. type(r))
end
elseif (argument == "say numeric") then
local r = set_skill(ch, "132", "83")
if (type(r) == "userdata" and r == ch.struct) then
say("numeric player struct")
else
say("numeric player result " .. type(r))
end
elseif (argument == "say mobile") then
local r = set_skill(me, 134, 37)
if (type(r) == "userdata" and r == me.struct) then
say("valid mobile struct")
else
say("valid mobile result " .. type(r))
end
elseif (argument == "say badchar") then
local r = set_skill("not a table", 134, 37)
say("invalid character top " .. r)
elseif (argument == "say badskill") then
local r = set_skill(ch, "not a number", 37, "top")
say("invalid skill top " .. r)
elseif (argument == "say badlevel") then
local r = set_skill(ch, 134, "not a number")
say("invalid level top " .. r)
elseif (argument == "say nil") then
local r = set_skill(ch, 134, nil)
say("invalid nil top " .. type(r))
elseif (argument == "say table") then
local r = set_skill(ch, 134, ch)
if (r == ch) then
say("invalid table top unchanged")
else
say("invalid table changed")
end
else
return FALSE
end
return TRUE
end
