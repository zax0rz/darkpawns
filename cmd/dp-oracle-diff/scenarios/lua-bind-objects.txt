[lua-script mob/oracle/probe.lua]
function oncmd()
local found, e, cmd, what = strfind(argument, "^(%a+) (%a+)")
if (cmd ~= "say") then
return FALSE
end
if (what == "mload") then
local m = mload(12204, 12296)
local none = mload(99999, 12296)
say("mload " .. m.name .. " vnum=" .. m.vnum .. " none=" .. tostring(none))
elseif (what == "list") then
local mine = obj_list("tooth", "char")
local theirs = obj_list("tooth", "vict")
say("obj_list char=" .. tostring(mine) .. " vict=" .. tostring(theirs) .. " obj=" .. obj.name .. " ch=" .. ch.name)
elseif (what == "steal") then
obj_list("tooth", "vict")
steal(ch, obj)
say("steal me has " .. getn(me.objs))
elseif (what == "back") then
local o = me.objs[1]
local r = objfrom(o, "char")
local r2 = objto(o, "char", ch)
say("objfrom/objto " .. type(r) .. " " .. type(r2))
elseif (what == "drop") then
local o = ch.objs[1]
objfrom(o, "char")
objto(o, "room", 12296)
say("dropped " .. o.name)
elseif (what == "equip") then
equip_char(ch, ch.objs[1])
say("equipped")
elseif (what == "extra") then
local r = extra(ch.objs[1], " Now with more hops.")
say("extra returned " .. type(r))
else
return FALSE
end
return TRUE
end
