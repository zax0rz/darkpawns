[lua-script mob/oracle/call_lightning.lua]
function oncmd()
if (argument == "say shield") then
aff_flags(ch, "set", 7)
aff_flags(ch, "set", 13)
return TRUE
end
if (argument == "say state") then
say("sanctuary=" .. tostring(aff_flagged(ch, 7)) .. " protect-good=" .. tostring(aff_flagged(ch, 13)))
return TRUE
end
return FALSE
end
