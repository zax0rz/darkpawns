# depth-case: lua.fight-npc-target-binding
# C-first vehicle for run_script's ch binding when the opponent is a mobile
# (DP-1319). The fixture gives the rat prototype a fight script: a charmed
# rat ordered to kill a mold makes hit() end in run_script(victim, ch=rat)
# with the MOLD as ch (fight.c:1889-1892), and C binds that mobile through
# char_to_table (scripts.c:1727-1729), so the act pair renders with $N = the
# mold every round. The unfixed Go dropped the mobile opponent: ch stayed
# unset, the act call errored on ch, and no line ever printed.

[fixture]
empty-players
quiet-mobs
replace-room-exits 10049 none
mob-script 4500 oracle/glm1319fight.lua 256

[lua-script mob/oracle/glm1319fight.lua]
function fight()
  act("$n sniffs at $N suspiciously!", TRUE, me, NIL, ch, TO_NOTVICT)
  act("$n sniffs at you suspiciously!", TRUE, me, NIL, ch, TO_VICT)
end
